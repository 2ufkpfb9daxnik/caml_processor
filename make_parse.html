4-2. 構文解析器の自作とパーサコンビネータ

本章までは ocamllex や ocamlyacc などのパーサジェネレータを利用して抽象構文木（AST）を構築する手法を前提としてきた。しかし、プログラム言語の処理系の全容を真に理解するためには、ツールに依存せず、構文解析の仕組みを自作（フルスクラッチで実装）することが極めて有効である。

ここでは、言語の機能を少しずつ拡張しながらパーサを実装していく（インクリメンタルな開発）。その過程で、愚直な実装が抱える問題点に直面し、それを「関数プログラミングならではの手法」でいかに美しく解決するかを体験してほしい。

ステップ1: 最小構成のASTと基本原理

構文解析（パース）の基本原理は、「入力文字列の先頭から意味のある塊（トークン）を読み取り、ASTの一部を作り、まだ読んでいない残りの文字列を次の処理へ渡す」ことの繰り返しである。
  • ミニ目的1-1: 最小のAST定義: まずは2-2や2-3で登場するような、整数と加算のみを表現するデータ型を定義する。
  • ミニ目的1-2: 入力と出力の型定義: 入力を char list とし、成功すれば Some (結果, 残りの文字リスト) を返す構造を作る。 
(* ミニ目的1-1: 最小のAST定義 *)
type exp =
  | EInt of int
  | EAdd of exp * exp

(* ミニ目的1-2: 入力と出力の型定義 *)
type input = char list
type 'a parser_result = ('a * input) option

ステップ2: 簡単な構文のパース（整数と加算）

まずは OCaml の基本的な機能である match 式を使って、愚直にパーサを実装してみよう。 （※ここでは int_p (整数を読む関数) や keyword_p (特定の文字列を読む関数) などの補助関数は既に存在するものとする）。
  • ミニ目的2-1: 加算のパース: 「左辺を読み、+ を読み、右辺を読む」という処理を match で繋いで記述する。 
(* ミニ目的2-1: 加算のパース（match を使った素直な実装） *)
let rec add_expr (input: input) : exp parser_result =
  (* 1. まず左辺（整数）を読む *)
  match int_p input with
  | None -> None
  | Some (e1, rest1) ->
      (* 2. 次に "+" 記号を読む *)
      match keyword_p "+" rest1 with
      | None -> Some (e1, rest1) (* "+" が無ければ、単なる整数として e1 を返す *)
      | Some (_, rest2) ->
          (* 3. "+" があれば、右辺（整数）を読む *)
          match int_p rest2 with
          | None -> None
          | Some (e2, rest3) -> 
              (* 4. 左右が揃ったので、EAddのASTを作って返す *)
              Some (EAdd (e1, e2), rest3)
この時点で、add_expr ['1'; '+'; '2'] のような入力を与えれば、正しく Some (EAdd (EInt 1, EInt 2), []) というASTを返すことができるパーサが完成している。状態（rest1, rest2）が順に渡っていく様子がよくわかるだろう。

ステップ3: 真理値と比較演算子の追加

次に、インクリメンタルな拡張として bool と比較演算子 = を追加してみよう。
  • ミニ目的3-1: ASTの拡張: EBool と EEq を exp 型に追加する。
  • ミニ目的3-2: 比較演算子のパース: 加算のパーサと同じ要領で、match を使って比較演算の処理を追加する。 
(* ミニ目的3-1: ASTの拡張 *)
type exp =
  | EInt of int
  | EBool of bool       (* 追加 *)
  | EAdd of exp * exp
  | EEq of exp * exp    (* 追加 *)

(* ミニ目的3-2: 比較演算子のパース *)
let rec comp_expr (input: input) : exp parser_result =
  (* 1. まず左辺（加算式など）を読む *)
  match add_expr input with
  | None -> None
  | Some (e1, rest1) ->
      (* 2. 次に "=" を読む *)
      match keyword_p "=" rest1 with
      | None -> Some (e1, rest1) (* "=" が無ければ、e1 だけを返す *)
      | Some (_, rest2) ->
          (* 3. 右辺を読む *)
          match add_expr rest2 with
          | None -> None
          | Some (e2, rest3) -> 
              Some (EEq (e1, e2), rest3)
少しコードが長くなってきたが、まだ match のネスト（入れ子）は2段階程度であり、処理の流れを十分に目で追うことができる。

ステップ4: if式の追加と直面する課題

いよいよ、制御構文の要である if e1 then e2 else e3 を追加する。これまでと同じ方針で match を繋げて書いてみよう。
  • ミニ目的4-1: if式のAST追加: EIf of exp * exp * exp を追加する。
  • ミニ目的4-2: if式のパースと課題の顕在化: if, e1, then, e2, else, e3 を順番に読んでいく。 
(* ミニ目的4-2: if式のパース（悪い例） *)
let rec if_expr (input: input) : exp parser_result =
  match keyword_p "if" input with
  | Some (_, rest1) ->
      (match comp_expr rest1 with
       | Some (e1, rest2) ->
           (match keyword_p "then" rest2 with
            | Some (_, rest3) ->
                (match comp_expr rest3 with
                 | Some (e2, rest4) ->
                     (match keyword_p "else" rest4 with
                      | Some (_, rest5) ->
                          (match comp_expr rest5 with
                           | Some (e3, rest6) -> 
                               Some (EIf (e1, e2, e3), rest6)
                           | None -> None)
                      | None -> None)
                 | None -> None)
            | None -> None)
       | None -> None)
  | None -> comp_expr input

問題 (提出課題ではない)

  • 上記の if_expr のコードを見て、問題点や保守しづらい理由を言葉で説明せよ。
実際に書いてみると一目瞭然だが、コードが右へ右へと深くネストしていく「コールバック地獄」に陥ってしまっている。 さらに深刻なのは、| None -> None という単なるエラー処理の分岐が大量に発生し、本来私たちが表現したかった if ... then ... else ... という文法規則の構造が、コードのノイズに完全に埋もれてしまっていることである。

ステップ5: 高階関数による解決（パーサコンビネータの導入）

ステップ4で発生した「同じようなパターンの連続」を観察してみよう。 行われているのは常に、「前のパースが成功したら、その結果（生成物と残りの文字列）を使って、次のパースを行う。失敗したら None を返す」という処理である。 関数プログラミングでは、このような「共通する処理フロー」をデザインパターンで解決するのではなく、関数そのものを抽象化してくくり出す（高階関数にする）ことで解決する。
  • ミニ目的5-1: 順次実行の関数 (bind) の定義: 処理を繋ぐ高階関数を定義する。 
(* 状態の受け渡しとエラー処理を隠蔽する高階関数 *)
let bind p f =
  fun input ->
    match p input with
    | None -> None
    | Some (v, rest) -> f v rest
この bind 関数を使えば、先ほどの深い match のネストをフラットに記述できるようになる。しかし、OCamlではこれをさらに読みやすくするため、独自の演算子（糖衣構文）として定義し直すことができる。
  • ミニ目的5-2: 中置演算子としての定義: bind を演算子 >>= として定義する。
  • ミニ目的5-3: if式の再実装: 演算子を用いて if_expr を書き直す。 
(* ミニ目的5-2: 演算子の定義 *)
let ( >>= ) = bind

(* 常に成功して値を返すだけの小さなパーサ *)
let return v = fun input -> Some (v, input)

(* ミニ目的5-3: 高階関数を用いてリファクタリングした if_expr *)
let rec if_expr input =
  (keyword_p "if"   >>= fun _  ->
   comp_expr        >>= fun e1 ->
   keyword_p "then" >>= fun _  ->
   comp_expr        >>= fun e2 ->
   keyword_p "else" >>= fun _  ->
   comp_expr        >>= fun e3 ->
   return (EIf (e1, e2, e3)))
  input
いかがだろうか。match と None の嵐が消え去り、文法の定義そのものと言えるほど劇的にコードが美しく、読みやすくなったはずだ。
このように、「小さなパーサ」を「高階関数（コンビネータ）」で組み合わせて複雑なパーサを構築していく手法をパーサコンビネータと呼ぶ。次節では、このコンビネータをさらに拡充し、本格的な言語処理系のパーサを完成させていく。
